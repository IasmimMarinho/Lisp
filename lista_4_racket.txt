(define (soma-impares lst)
  (define (aux lst ac)
    (cond
      [(null? lst) ac]                                  
      [(odd? (first lst)) (aux (rest lst) (+ ac (first lst)))]
      [else (aux (rest lst) ac)]))                        
  (aux lst 0))

;Q2

(define (produto-elementos lst)
  (define (aux lst ac)
    (cond
      [(null? lst) ac]                                  
      [else (aux (rest lst) (* ac (first lst)))]))                        
  (aux lst 1))

(define (substitui* old new lst)
  (cond 
   [(null? lst) '()]
   [(list? (first lst))
    (cons (substitui* old new (first lst))
    	  (substitui* old new (rest lst)))]
   [(eqv? old (first lst))
    (cons new 
	(substitui* old new (rest lst)))]
   [else
    (cons (first lst)
	(substitui* old new (rest lst)))]
))

(define (inverte* lg)
  (inv-aux* lg '()))

(define (inv-aux* lg lga)
  (cond
    [(null? lg) lga]
    [(list? (first lg))
     (inv-aux* (rest lg) (cons (inv-aux* (first lg) '()) lga))]
    [else
     (inv-aux* (rest lg) (cons (first lg) lga))]
    ))

(define (rmembro* a lg)
  (rest (r-uma-aux a lg)))
  
(define (r-uma-aux a lg)
  (cond
    [(null? lg) (list #f)]
    [(list? (first lg))
       (let [(resp (r-uma-aux a (first lg)))]
         (if (first resp)
             (cons #t
                   (cons (rest resp)
                         (rest lg)))
             (let [(resp2 (r-uma-aux a (rest lg)))]
               (cons (first resp2)
                     (cons (first lg)
                           (rest resp2))))))]
    [(eqv? a (first lg))
       (cons #t (rest lg))]
    [else
       (let [(resp2 (r-uma-aux a (rest lg)))]
               (cons (first resp2)
                     (cons (first lg)
                           (rest resp2))))]
    ))
